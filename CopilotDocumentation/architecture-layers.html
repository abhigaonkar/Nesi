🔐 Authentication Service

Responsibility: Manage OAuth tokens, login/logout, token refresh

export class AuthService { private tokenKey = 'auth_token'; login(username: string, password: string): Observable { const body = `grant_type=password&username=${username}&password=${password}`; return this.http.post('/token', body, { headers: { 'Content-Type': 'application/x-www-form-urlencoded' } }).pipe( tap(response => this.storeToken(response.access_token)), catchError(this.handleAuthError) ); } getToken(): string { return localStorage.getItem(this.tokenKey); } isAuthenticated(): boolean { return !!this.getToken() && !this.isTokenExpired(); } }

🔌 HTTP Interceptor

Responsibility: Automatically attach Bearer token to API requests

@Injectable() export class AuthInterceptor implements HttpInterceptor { constructor(private authService: AuthService) {} intercept(req: HttpRequest, next: HttpHandler): Observable> { const token = this.authService.getToken(); if (token) { req = req.clone({ setHeaders: { Authorization: `Bearer ${token}` } }); } return next.handle(req).pipe( catchError((error: HttpErrorResponse) => { if (error.status === 401) { this.authService.logout(); this.router.navigate(['/login']); } return throwError(error); }) ); } }

📊 State Management (@ngrx/store)

Responsibility: Centralized application state with Redux pattern

// Customer State Interface export interface CustomerState { customers: Customer[]; selectedCustomer: Customer | null; loading: boolean; error: string | null; } // Customer Actions export const loadCustomers = createAction('[Customer] Load Customers'); export const loadCustomersSuccess = createAction( '[Customer] Load Customers Success', props<{ customers: Customer[] }>() ); // Customer Reducer export const customerReducer = createReducer( initialState, on(loadCustomers, state => ({ ...state, loading: true })), on(loadCustomersSuccess, (state, { customers }) => ({ ...state, customers, loading: false })) ); // Customer Effects @Injectable() export class CustomerEffects { loadCustomers$ = createEffect(() => this.actions$.pipe( ofType(loadCustomers), switchMap(() => this.customerService.getCustomers().pipe( map(customers => loadCustomersSuccess({ customers })), catchError(error => of(loadCustomersFailure({ error }))) ) ) ) ); }

namespace NESI.WebAPI.Controllers.API.Page.Customers { [RoutePrefix("api/page/customers")] [PageAuthorizationFilter(PageId = 15)] // Page-level auth public class CustomerController : EmployeeController { private readonly ICustomerService _customerService; public CustomerController(ICustomerService customerService) { _customerService = customerService; } [HttpGet] [Route("search")] public IHttpActionResult Search(string query) { try { var customers = _customerService.Search( query, CurrentUser.BusinessUnitIds // Auto-filtered by BU ); return Ok(customers); } catch (Exception ex) { Logger.Error("Customer search failed", ex); return InternalServerError(ex); } } [HttpPost] [Route("")] [PrivilegeAuthorizationFilter(PrivilegeId = 101)] // Create privilege public IHttpActionResult Create(CustomerDto customer) { if (!ModelState.IsValid) return BadRequest(ModelState); try { var result = _customerService.Create(customer, CurrentUser.Id); return Ok(result); } catch (BusinessException bex) { return BadRequest(bex.Message); } } } }

Custom Authorization Filters

PageAuthorizationFilterAttribute

public class PageAuthorizationFilterAttribute : NesiFilterAttributeBase { protected int PageId { get; set; } public PageAuthorizationFilterAttribute(int pageId) { PageId = pageId; } protected override bool IsAuthorized(HttpActionContext actionContext) { var user = base.GetCurrentUser(actionContext); return user != null && user.AuthorizePage(PageId); } }

PrivilegeAuthorizationFilterAttribute

public class PrivilegeAuthorizationFilterAttribute : NesiFilterAttributeBase { protected int PrivilegeId { get; set; } public PrivilegeAuthorizationFilterAttribute(int privilegeId) { PrivilegeId = privilegeId; } protected override bool IsAuthorized(HttpActionContext actionContext) { var user = base.GetCurrentUser(actionContext); return user != null && user.AuthorizePrivilege(PrivilegeId); } }

Service Layer Example

public class QuoteService : IQuoteService { private readonly NESIEntities _context; private readonly IMapper _mapper; public QuoteService(NESIEntities context, IMapper mapper) { _context = context; _mapper = mapper; } public QuoteDto CreateQuote(QuoteDto quoteDto, int userId) { // Business validation ValidateQuoteData(quoteDto); // Business logic var quote = _mapper.Map(quoteDto); quote.CreatedBy = userId; quote.CreatedDate = DateTime.Now; quote.Status = QuoteStatus.Draft; // Calculate totals CalculateQuoteTotals(quote); // Save to database _context.Quotes.Add(quote); _context.SaveChanges(); // Return result return _mapper.Map(quote); } private void CalculateQuoteTotals(Quote quote) { quote.SubTotal = quote.QuoteLines.Sum(l => l.Quantity * l.UnitPrice); quote.TaxAmount = quote.SubTotal * quote.TaxRate; quote.Total = quote.SubTotal + quote.TaxAmount; } public WorkOrderDto ConvertToWorkOrder(int quoteId, int userId) { var quote = _context.Quotes .Include(q => q.QuoteLines) .Include(q => q.Customer) .FirstOrDefault(q => q.Id == quoteId); if (quote == null) throw new BusinessException("Quote not found"); if (quote.Status != QuoteStatus.Approved) throw new BusinessException("Only approved quotes can be converted"); // Create work order from quote var workOrder = new WorkOrder { QuoteId = quote.Id, CustomerId = quote.CustomerId, Description = quote.Description, EstimatedStartDate = quote.EstimatedStartDate, EstimatedEndDate = quote.EstimatedEndDate, Status = WorkOrderStatus.Open, CreatedBy = userId, CreatedDate = DateTime.Now }; // Copy line items foreach (var quoteLine in quote.QuoteLines) { workOrder.WorkOrderLines.Add(new WorkOrderLine { Description = quoteLine.Description, Quantity = quoteLine.Quantity, UnitPrice = quoteLine.UnitPrice, Total = quoteLine.Total }); } _context.WorkOrders.Add(workOrder); quote.Status = QuoteStatus.Converted; _context.SaveChanges(); return _mapper.Map(workOrder); } }

Caching Strategy

public class CacheManager { private static MemoryCache _cache = MemoryCache.Default; public static T Get(string key, Func valueFactory, TimeSpan? expiration = null) { if (_cache.Contains(key)) return (T)_cache.Get(key); var value = valueFactory(); var policy = new CacheItemPolicy { AbsoluteExpiration = DateTimeOffset.Now.Add( expiration ?? TimeSpan.FromHours(1) ) }; _cache.Set(key, value, policy); return value; } public static void Remove(string key) { _cache.Remove(key); } } // Usage var businessUnits = CacheManager.Get( "BusinessUnits", () => _context.BusinessUnits.ToList(), TimeSpan.FromHours(24) );
